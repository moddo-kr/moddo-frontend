{/* 컬러 스와치 안의 텍스트의 색상을 결정하기 위한 함수 */}
export const isLight = (hex) => {
  const r = parseInt(hex.slice(1, 3), 16);
  const g = parseInt(hex.slice(3, 5), 16);
  const b = parseInt(hex.slice(5, 7), 16);
  return (0.299 * r + 0.587 * g + 0.114 * b) / 255 > 0.55;
};

export const ColorStrip = ({ entries }) => (
  <div style={{ display: 'flex', height: 56, borderRadius: 4, overflow: 'hidden', marginBottom: 8 }}>
    {entries.map(([key, hex]) => {
      const step = key.split('.')[1];
      const light = isLight(hex);
      return (
        <div
          key={key}
          title={`${key}: ${hex}`}
          style={{
            flex: 1,
            minWidth: 0,
            backgroundColor: hex,
            display: 'flex',
            flexDirection: 'column',
            alignItems: 'center',
            justifyContent: 'center',
            gap: 2,
            color: light ? '#292c30' : '#ffffff',
          }}
        >
          <span style={{ fontSize: 11, fontFamily: 'monospace', fontWeight: 600 }}>{step}</span>
          <span style={{ fontSize: 9, fontFamily: 'monospace', opacity: 0.7 }}>{hex}</span>
        </div>
      );
    })}
  </div>
);

export const commonEntries = Object.entries(atomicColorTokens)
  .filter(([k]) => k.startsWith('common.'));

export const grayEntries = Object.entries(atomicColorTokens)
  .filter(([k]) => k.startsWith('gray.') && k !== 'gray.0' && k !== 'gray.100')
  .reverse();

export const coolGrayEntries = Object.entries(atomicColorTokens)
  .filter(([k]) => k.startsWith('coolGray.') && k !== 'coolGray.0')
  .reverse();

export const orangeEntries = Object.entries(atomicColorTokens)
  .filter(([k]) => k.startsWith('orange.'))
  .reverse();

export const yellowEntries = Object.entries(atomicColorTokens)
  .filter(([k]) => k.startsWith('yellow.'))
  .reverse();

export const redEntries = Object.entries(atomicColorTokens)
  .filter(([k]) => k.startsWith('red.'))
  .reverse();

export const greenEntries = Object.entries(atomicColorTokens)
  .filter(([k]) => k.startsWith('green.'))
  .reverse();

export const blueEntries = Object.entries(atomicColorTokens)
  .filter(([k]) => k.startsWith('blue.'))
  .reverse();

# Atomic Color

색상은 정보 전달과 시각적 경험을 결정하는 핵심 요소로, 사용자의 주의를 유도하고 메시지를 명확히 전달하여 직관적인 사용자 경험을 형성한다. Atomic Color는 디자인 시스템의 가장 기초 단위로, 모든 Semantic 색상 토큰의 기준이 되는 원시 색상 값으로 구성된다. 이를 통해 일관된 색상 체계와 시각적 계층 구조를 유지한다.

---

## 공통 (Common)

Common 색상은 서비스 전반에서 기본적으로 반복 사용되는 핵심 색상(Black, White 등) 으로, 배경이나 텍스트 등 모든 테마에서 공통적으로 유지되는 기준 색상이다.
