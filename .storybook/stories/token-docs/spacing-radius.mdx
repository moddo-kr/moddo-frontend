## Shadow

<table style={tableStyle}>
  <thead>
    <tr>
      <th style={thStyle}>Token</th>
      <th style={thStyle}>Value</th>
      <th style={thStyle}>CSS Variable</th>
      <th style={thStyle}>Preview</th>
    </tr>
  </thead>
  <tbody>
    {Object.entries(atomicShadowTokens).map(([key, { x, y, blur, spread, color }]) => {
      const cssVar = `--${key.replace(/shadow(\d+)/, 'shadow-$1')}`;
      const value = `${x}px ${y}px ${blur}px ${spread}px ${color}`;
      return (
        <tr key={key}>
          <td style={tdStyle}><code>{key}</code></td>
          <td style={tdStyle}><code>{value}</code></td>
          <td style={tdStyle}><code>{cssVar}</code></td>
          <td style={tdStyle}>
            <div style={{
              width: 48,
              height: 48,
              backgroundColor: 'var(--color-fill-normal)',
              border: '1px solid var(--color-border-alternative)',
              borderRadius: 'var(--radius-md)',
              boxShadow: `var(${cssVar})`,
            }} />
          </td>
        </tr>
      );
    })}
  </tbody>
</table>
